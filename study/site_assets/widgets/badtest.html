Spot the bad test

An unsolved stub raises NotImplementedError. A good test makes that show as TODO. Which of these tests would give a wrong result on a stub?

def test_rejects_none():
    with pytest.raises(Exception):
        ex.parse(None)
def test_budget_exceeded():
    with pytest.raises(RuntimeError):
        tracker.spend(10**9)
def test_needs_two_args():
    with pytest.raises(TypeError):
        ex.take(1, 2, 3)
def test_take_stops_early():
    it = iter(range(10))
    assert list(ex.take(3, it)) == [0, 1, 2]
    assert next(it) == 3
